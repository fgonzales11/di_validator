cmake_minimum_required(VERSION 3.10)
project(PVDetectionCore LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(pv_core core/pv_core.cpp core/pv_state.cpp core/pv_wire.cpp core/pv_delivery.cpp)
target_include_directories(pv_core PUBLIC core)
target_compile_options(pv_core PRIVATE -Wall -Wextra -Wconversion -Werror)
add_executable(pv_replay host/pv_replay.cpp host/replay_io.cpp)
target_include_directories(pv_replay PRIVATE host)
target_link_libraries(pv_replay PRIVATE pv_core)
enable_testing()
add_executable(pv_tests tests/core_tests.cpp)
target_link_libraries(pv_tests PRIVATE pv_core)
add_test(NAME pv_core_contracts COMMAND pv_tests)
