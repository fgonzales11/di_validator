cmake_minimum_required(VERSION 3.10)
project(FaultLocationCore LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(fault_core core/fault_core.cpp core/fault_wire.cpp)
target_include_directories(fault_core PUBLIC core)
target_compile_options(fault_core PRIVATE -Wall -Wextra -Wconversion -Werror)
add_executable(fault_replay host/replay_main.cpp host/replay_io.cpp)
target_include_directories(fault_replay PRIVATE host)
target_link_libraries(fault_replay PRIVATE fault_core)
target_compile_options(fault_replay PRIVATE -Wall -Wextra -Wconversion -Werror)
enable_testing()
add_executable(fault_kernel_tests tests/kernel_tests.cpp)
target_link_libraries(fault_kernel_tests PRIVATE fault_core)
target_compile_options(fault_kernel_tests PRIVATE -Wall -Wextra -Wconversion -Werror)
add_test(NAME fault_kernel_tests COMMAND fault_kernel_tests)
